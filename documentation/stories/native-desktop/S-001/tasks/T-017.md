Implement the first-run empty state in the contact sheet area: a centered drop zone that accepts dragged folders as the 'Choose original folder' and 'Find copies in' sources. ## How

**Step 1 — Create EmptyState component:**
File: `src/components/EmptyState.tsx`
```tsx
import React from 'react';
import './EmptyState.css';

interface EmptyStateProps {
  title: string;
  body?: string;
  action?: { label: string; onClick: () => void };
  dropZone?: {
    onDrop: (paths: string[]) => void;
    hint?: string;
  };
}

export function EmptyState({ title, body, action, dropZone }: EmptyStateProps) {
  const [isDragging, setIsDragging] = React.useState(false);

  const handleDrop = dropZone ? async (e: React.DragEvent) => {
    e.preventDefault();
    setIsDragging(false);
    // Tauri 2: files dropped via the webview come through the file drop event
    // The paths are available via e.dataTransfer.files
    const paths = Array.from(e.dataTransfer.files).map(f => f.path);
    if (paths.length > 0) dropZone.onDrop(paths);
  } : undefined;

  return (
    <div
      className='empty-state'
      data-drop-zone={!!dropZone}
      data-dragging={isDragging}
      onDragOver={dropZone ? e => { e.preventDefault(); setIsDragging(true); } : undefined}
      onDragLeave={dropZone ? () => setIsDragging(false) : undefined}
      onDrop={handleDrop}
    >
      <h2 className='empty-state-title'>{title}</h2>
      {body && <p className='empty-state-body'>{body}</p>}
      {dropZone?.hint && <p className='empty-state-hint'>{dropZone.hint}</p>}
      {action && (
        <button className='empty-state-action' onClick={action.onClick}>
          {action.label}
        </button>
      )}
    </div>
  );
}
```

**Step 2 — Update App.tsx to show first-run empty state:**
```tsx
function App() {
  const [folders, setFolders] = useState<{ original?: string; copies?: string }>({});

  const handleDrop = (paths: string[]) => {
    // First drop = original folder; second drop = copies folder
    if (!folders.original) setFolders(f => ({ ...f, original: paths[0] }));
    else setFolders(f => ({ ...f, copies: paths[0] }));
  };

  const hasNoFolders = !folders.original;

  return (
    <AppShell
      contactSheet={
        hasNoFolders ? (
          <EmptyState
            title='Drop a folder to get started'
            body='Drop your originals folder, then your copies folder, to find duplicates.'
            dropZone={{ onDrop: handleDrop, hint: 'or click Choose Folder in the sidebar' }}
          />
        ) : (
          <div>Scanning... (contact sheet coming)</div>
        )
      }
      // ...
    />
  );
}
```
