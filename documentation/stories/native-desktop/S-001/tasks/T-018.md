# native-desktop/S-001-T18: Keyboard shortcuts — 11 bindings per DD7

**Status:** done
**Priority:** P2
**Blocked By:** S-001-T14

## What
Register all 11 keyboard shortcuts from design decision DD7 using a useKeyboardShortcuts hook. Wire Cmd+F to focus the search bar, Space to open Quick Look (or inspector), arrow keys to navigate the contact sheet, Enter to confirm quarantine, Escape to cancel/close, A to select all, Delete to quarantine selected, Cmd+Z to undo last quarantine.

## Why
Design decision DD7: '11 keyboard shortcuts.' The Python Streamlit app requires clicking every action. For a reviewer processing 500 duplicate groups, keyboard-driven review (arrow to next, Enter to quarantine, Cmd+Z to undo) is 10x faster than mouse-only.

## Context
Full keyboard shortcut list (DD7):
1. Cmd+F — focus search bar
2. Space — open Quick Look preview for focused image
3. ArrowRight / ArrowDown — navigate to next group
4. ArrowLeft / ArrowUp — navigate to previous group
5. Enter — confirm quarantine of selected copies
6. Escape — cancel selection / close inspector / clear search
7. A (no modifier) — select all copies in focused group
8. Delete / Backspace — add selected to quarantine queue (same as quarantine button)
9. Cmd+Z — undo last quarantine (triggers post-quarantine banner Undo)
10. Cmd+A — select all groups
11. Tab — cycle focus between sidebar / contact sheet / inspector

Implementation: a useKeyboardShortcuts React hook that adds window-level keydown listeners. The hook is mounted once in App.tsx.

## How

**Step 1 — Create useKeyboardShortcuts hook:**
File: `src/hooks/useKeyboardShortcuts.ts`
```tsx
import { useEffect } from 'react';

export interface KeyboardShortcutHandlers {
  onSearch?: () => void;         // Cmd+F
  onPreview?: () => void;        // Space
  onNavigateNext?: () => void;   // ArrowRight / ArrowDown
  onNavigatePrev?: () => void;   // ArrowLeft / ArrowUp
  onConfirm?: () => void;        // Enter
  onCancel?: () => void;         // Escape
  onSelectAll?: () => void;      // A (no mod) / Cmd+A
  onQuarantine?: () => void;     // Delete / Backspace
  onUndo?: () => void;           // Cmd+Z
}

export function useKeyboardShortcuts(handlers: KeyboardShortcutHandlers) {
  useEffect(() => {
    const handle = (e: KeyboardEvent) => {
      // Don't steal shortcuts when user is typing in an input
      const target = e.target as HTMLElement;
      const isInput = target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable;

      const cmd = e.metaKey || e.ctrlKey;

      if (cmd && e.key === 'f') { e.preventDefault(); handlers.onSearch?.(); return; }
      if (cmd && e.key === 'z') { e.preventDefault(); handlers.onUndo?.(); return; }
      if (cmd && e.key === 'a') { e.preventDefault(); handlers.onSelectAll?.(); return; }

      if (isInput) return; // only non-input shortcuts below

      if (e.key === ' ') { e.preventDefault(); handlers.onPreview?.(); return; }
      if (e.key === 'ArrowRight' || e.key === 'ArrowDown') { e.preventDefault(); handlers.onNavigateNext?.(); return; }
      if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') { e.preventDefault(); handlers.onNavigatePrev?.(); return; }
      if (e.key === 'Enter') { e.preventDefault(); handlers.onConfirm?.(); return; }
      if (e.key === 'Escape') { e.preventDefault(); handlers.onCancel?.(); return; }
      if (e.key === 'a' && !cmd) { handlers.onSelectAll?.(); return; }
      if (e.key === 'Delete' || e.key === 'Backspace') { e.preventDefault(); handlers.onQuarantine?.(); return; }
    };

    window.addEventListener('keydown', handle);
    return () => window.removeEventListener('keydown', handle);
  }, [handlers]);
}
```

**Step 2 — Wire up hook in App.tsx:**
```tsx
const searchRef = useRef<HTMLInputElement>(null);

useKeyboardShortcuts({
  onSearch: () => searchRef.current?.focus(),
  onCancel: () => {
    if (document.activeElement === searchRef.current) {
      setSearchQuery('');
      searchRef.current?.blur();
    }
    // else: close inspector / deselect
  },

```

**Step 3 — Write unit tests:**
File: `src/hooks/useKeyboardShortcuts.test.ts`
```tsx
import { renderHook } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { useKeyboardShortcuts } from './useKeyboardShortcuts';

test('Cmd+F calls onSearch', async () => {
  const onSearch = vi.fn();
  renderHook(() => useKeyboardShortcuts({ onSearch }));
  await userEvent.keyboard('{Meta>}f{/Meta}');
  expect(onSearch).toHaveBeenCalledTimes(1);
});

test('Space calls onPreview (not in input)', async () => {
  const onPreview = vi.fn();
  renderHook(() => useKeyboardShortcuts({ onPreview }));
  await userEvent.keyboard(' ');
  expect(onPreview).toHaveBeenCalledTimes(1);
});

test('Space does NOT fire when input is focused', async () => {
  const onPreview = vi.fn();
  renderHook(() => useKeyboardShortcuts({ onPreview }));
  const input = document.createElement('input');
  document.body.appendChild(input);
  input.focus();
  await userEvent.keyboard(' ');
  expect(onPreview).not.toHaveBeenCalled();
  document.body.removeChild(input);
});
```
